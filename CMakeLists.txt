cmake_minimum_required(VERSION 3.20)
project(RESONIQ VERSION 0.2.16 LANGUAGES CXX)
configure_file(apps/resoniq_version.h.in generated/resoniq_version.h @ONLY)
option(RESONIQ_STATIC_RUNTIME "Use the static MSVC runtime for self-contained installers" OFF)
if(MSVC AND RESONIQ_STATIC_RUNTIME)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Shared diagnostics and floating-point contract for every owned target.
add_library(resoniq_options INTERFACE)
if(MSVC)
  target_compile_options(resoniq_options INTERFACE /W4 /permissive- /utf-8 /fp:precise)
else()
  target_compile_options(resoniq_options INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_library(dps src/dsp/engine.cpp src/dsp/measure.cpp
  src/io/wav.cpp src/io/profile.cpp src/io/preferences.cpp src/io/calibration_suggestion.cpp)
target_include_directories(dps PUBLIC include)
target_link_libraries(dps PRIVATE resoniq_options)

function(resoniq_executable target source)
  add_executable(${target} ${source})
  target_link_libraries(${target} PRIVATE dps resoniq_options)
endfunction()
resoniq_executable(dps_wav apps/wav_main.cpp)
resoniq_executable(dps_benchmark apps/benchmark_main.cpp)
resoniq_executable(dps_quality apps/quality_main.cpp)
if(WIN32)
  add_library(resoniq_close SHARED installer/close_application.cpp)
  target_link_libraries(resoniq_close PRIVATE resoniq_options msi user32)
  target_compile_definitions(resoniq_close PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  add_executable(resoniq_device installer/device_main.cpp)
  target_link_libraries(resoniq_device PRIVATE resoniq_options setupapi newdev cfgmgr32 wintrust crypt32 advapi32)
  target_compile_definitions(resoniq_device PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  # Compile the Windows audio adapter once for both front ends.
  add_library(dps_windows src/audio/wasapi.cpp src/io/hardware.cpp)
  target_link_libraries(dps_windows PUBLIC dps ole32 uuid avrt advapi32 PRIVATE resoniq_options)
  target_compile_definitions(dps_windows PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  resoniq_executable(dps_live apps/live_main.cpp)
  target_link_libraries(dps_live PRIVATE dps_windows)
  enable_language(RC)
  add_executable(dps_gui WIN32 apps/gui_main.cpp apps/gui_theme.cpp apps/gui_icons.cpp apps/resoniq.rc)
  set_target_properties(dps_gui PROPERTIES OUTPUT_NAME resoniq)
  target_include_directories(dps_gui PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(dps_gui PRIVATE dps_windows resoniq_options
    comctl32 comdlg32 shell32 dwmapi gdiplus uxtheme)
  target_compile_definitions(dps_gui PRIVATE UNICODE _UNICODE)
endif()

include(CTest)
if(BUILD_TESTING)
  function(resoniq_test target source name)
    resoniq_executable(${target} ${source})
    add_test(NAME ${name} COMMAND ${target})
    set_tests_properties(${name} PROPERTIES TIMEOUT 60)
  endfunction()
  resoniq_test(dps_tests tests/tests.cpp dsp_validation)
  resoniq_test(dps_profile_tests tests/profiles.cpp profile_validation)
  resoniq_test(dps_comparison_tests tests/comparison.cpp comparison_validation)
  resoniq_test(dps_tonal_tests tests/tonal.cpp tonal_validation)
  resoniq_test(dps_transient_tests tests/transient.cpp transient_validation)
  resoniq_test(dps_measure_tests tests/measure.cpp measure_validation)
  resoniq_test(dps_intensity_tests tests/intensity.cpp intensity_validation)
  resoniq_test(dps_attack_tests tests/attack.cpp attack_validation)
  resoniq_test(dps_recovery_tests tests/recovery.cpp recovery_validation)
  resoniq_test(dps_virtual_input_tests tests/virtual_inputs.cpp virtual_input_validation)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME classic_audio_contract COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/tests/classic_audio_contract.py)
  resoniq_test(dps_preferences_tests tests/preferences.cpp preferences_validation)
  resoniq_test(dps_hardware_tests tests/hardware.cpp hardware_validation)
  resoniq_test(dps_wav_tests tests/wav.cpp wav_validation)
  if(WIN32)
    resoniq_test(resoniq_startup_tests tests/windows_startup.cpp windows_startup_validation)
    target_link_libraries(resoniq_startup_tests PRIVATE advapi32 shell32 ole32 uuid)

    target_compile_definitions(resoniq_shutdown_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_test(NAME device_preflight COMMAND resoniq_device --check)
    add_test(NAME gui_controls COMMAND dps_gui --self-test)
    set_tests_properties(gui_controls PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
    resoniq_test(dps_wasapi_tests tests/wasapi.cpp wasapi_error_validation)
    target_link_libraries(dps_wasapi_tests PRIVATE dps_windows)
    resoniq_test(dps_silence_tests tests/endpoint_silence.cpp endpoint_silence_validation)
    target_link_libraries(dps_silence_tests PRIVATE dps_windows)
  endif()
